jpx.addPlugin(
{
    myNoise      : function()
    {
        return this.forXY2(function(x,y,data)
        {
            // keep the alpha channel as is
            for(var i = 0; i < data.length-1; ++i)
            {
                var val = data[i];
                var min = val / 2;
                var max = val + min;
                var rand = Math.random();
                data[i] = min + (max-min) * rand;
            }
        });
    }
});
